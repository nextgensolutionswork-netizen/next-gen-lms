import {beforeEach, expect, it, vi} from 'vitest';
const m=vi.hoisted(()=>({insert:vi.fn(),update:vi.fn(),select:vi.fn(),audit:vi.fn()}));
vi.mock('@/lib/services/data-store',()=>({store:{leads:[],courses:[],users:[],persist:vi.fn()}}));
vi.mock('@/lib/services/audit-service',()=>({recordAuditLog:m.audit}));
vi.mock('@/lib/supabase/db',()=>({isLiveSupabaseEnabled:()=>true,createClient:()=>({from:()=>({insert:m.insert,update:()=>({eq:m.update}),select:()=>({order:m.select})})})}));
import {createLead,updateLeadStage,getLeads} from '@/lib/services/crm-service';
import {store} from '@/lib/services/data-store';
const id='11111111-1111-4111-8111-111111111111',actor='22222222-2222-4222-8222-222222222222';
beforeEach(()=>{vi.clearAllMocks();store.leads=[];m.insert.mockResolvedValue({error:null});m.update.mockResolvedValue({error:{code:'42501',message:'Synthetic RLS rejection'}});});
it('must reject a failed authoritative stage update',async()=>{
 store.leads.push({id,stage:'New',full_name:'Synthetic',notes:'',created_at:new Date().toISOString()} as any);
 await expect(updateLeadStage(id,'Follow-up','Synthetic note',actor,'2026-11-01')).rejects.toThrow();
});
it('must reject empty required course rather than insert null',async()=>{
 await expect(createLead({full_name:'Synthetic',phone:'9990001234',email:'audit@example.invalid',stage:'New',interested_course_id:'',counsellor_id:actor} as any)).rejects.toThrow();
});
it('must apply search to live query results',async()=>{
 m.select.mockResolvedValue({error:null,data:[{id,full_name:'Synthetic',stage:'New'}]});
 expect(await getLeads({search:'does-not-match'})).toEqual([]);
});
